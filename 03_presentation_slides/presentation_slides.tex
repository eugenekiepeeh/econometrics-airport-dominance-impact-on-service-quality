\begin{frame}{Motivation} % 
    \begin{itemize}
        \item \textbf{Benefits and Cost of Competition} –  Is competition always good and Can concentration be bad?%
        \vspace{0.2cm}
        
        \item \textbf{Primary Question} - How does firm behavior shift when they become the dominant option?
        \vspace{0.2cm}
        \normalsize
      

    \end{itemize}

\begin{frame}{Past Research Findings}
    \begin{itemize}
        \item \textbf{Diana, 2016} Dominance enables connectivity but introduces systemic delays (3.0--3.3 pp lower on-time rates).
        \item \textbf{Calzada, 2023 } Dominant carriers may "internalize" congestion, scheduling conservatively to protect network integrity.
        \item \textbf{Chen, 2013} Quality often declines following mergers that increase dominance and reduce pressure
        \item \textbf{Ciliberto, 2010} Monopoly status can shift focus from satisfaction to cost minimization \cite{ciliberto2010}.
        \item \textbf{Gans, 2017} Complaint volume is \textbf{twice as sensitive} to delays when an airline controls $>50\%$ of a market.
    \end{itemize}
\end{frame}

\begin{frame}{Research Gap}
    \begin{itemize}
        \item \textbf{On-time is the only metric that has been used in the past to assess service}
            \begin{itemize}
                \item This paper introduces more metrics that consumers care about: delay, arrival and cancellations.
            \end{itemize}
        \item \textbf{Limited studies (No study**) have been conducted on COVID and the post-COVID effect.}
            \begin{itemize}
                \item This paper presents the first multi year study on how market concentration impact service quality.
            \end{itemize}
    \end{itemize}
\end{frame}

% --- Methodology ---
\begin{frame}{Methodology} % 

    \begin{itemize}
        \item \textbf{Dependent Variable ($Y$)}
        \begin{itemize}
            \item \textbf{Departure Delay Rate:} Proportion of monthly flights departing $\geq 15$ mins late.
            \item \textbf{Arrival Delay Rate:} Proportion of monthly flights arriving later than $\geq 15$ mins.
            \item \textbf{Cancellation Rate:} Proportion of monthly flights canceled.
        \end{itemize}
        \addlinespace
        \item \textbf{Key Structural Predictors}
        \begin{itemize}
            \item \textbf{Market Concentration (HHI):} Sum of squared route market shares.
            \item \textbf{Centered Hub Share:} Carrier share at origin, centered on monthly airport mean to mitigate structural multicollinearity.
            \item \textbf{Schedule Dispersion ($sd\_dep\_time$):} Proxies for daily schedule volatility (scaled to 60-min units).
        \end{itemize}
    \end{itemize}
\end{frame}

% --- Theoretical background ---
\begin{frame}{Theoretical background} % 
    \begin{itemize}
        \item Model
    \end{itemize}
    \begin{equation}
    Y_{irt} = \beta_1 HHI_{rt} + \beta_2 Share_{irt} + \gamma X_{irt} + \alpha_i + \delta_r + \tau_t + \epsilon_{irt}
    \end{equation}
    \begin{itemize}
        \item where 
        \begin{itemize}
            \item $Y_{irt}$ represents the delay or cancellation rate for operating carrier $i$ on route $r$ in month-year $t$.
            \item $HHI_{rt}$ represents route concentration
            \item $Share_{irt}$ is the carrier's route market share
            \item $X_{irt}$ represents a vector of operational controls.

        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Econometric Specification}
    \begin{block}{Empirical Model: Fixed Effects OLS}
        \footnotesize
        \[ Delay_{icrt} = \beta_1 HHI_{rt} + \beta_2 Share_{icrt} + \beta_3 (HHI \times Hub)_{ict} + \beta_4 (HHI \times \ln Density)_{rt} + \dots \]
        \[ \dots + \alpha_c + \gamma_r + \delta_t + \epsilon_{icrt} \]
    \end{block}

    \begin{itemize}
        \item \textbf{Fixed Effects:}
        \begin{itemize}
            \item \textbf{Route ($\gamma_r$):} Controls for time-invariant geography and airport constraints.
            \item \textbf{Carrier ($\alpha_c$):} Accounts for airline-specific operational culture.
            \item \textbf{Year-Month ($\delta_t$):} Controls for macroeconomic and seasonal shocks.
        \end{itemize}
        \addlinespace
        \item \textbf{Identification:} Relies on \textit{within-variation} of HHI on specific corridors across 72 monthly time-steps.
        \item \textbf{Robustness:}
        \begin{itemize}
            \item \textbf{Clustering:} Standard errors clustered at the \textbf{Route level}.
            \item \textbf{Weighting:} Weighted by \textit{carrier total flights} to reflect the average passenger experience.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Analysis}
    \small
    \textbf{Part I: Key Variable Interpretations}
    \begin{itemize}
        \item \textbf{The Hub Penalty:} Airport dominance ($HubShare$) is the strongest predictor of delays. A 10\% increase in hub share increases baseline departure delays by \textbf{0.97 percentage points}.
        \item \textbf{Operational Efficiency:} Higher route share is a associated with lower arrival delays ($\beta = -0.052$).
        \item \textbf{(Interaction):} Airlines with large shares of departing flights on a highly concentrated routes are expected to reduced their delay by  0.0271 percentage point. 
        \begin{itemize}
            \item The same cannot be said for Cancellation rate! Higher route share shows an expected increase in delays by the dominant carrier.
        \end{itemize}
        
        
    \end{itemize}

    \hrule

\begin{frame}{Conclusion}

    \textbf{Does airport dominance weaken quality competition?}
    \vspace{0.1cm}
    \textbf{\color{red} Answer: Not Exactly Yes!}
    \begin{itemize}
        \item \textbf{Delays (Efficiency):} Dominance \textit{improves} reliability in concentrated markets. Dominant hub carriers "internalize" congestion, coordinating schedules better when competition is low to avoid self-congestion.
        \item \textbf{Cancellations (Strategic):} \textbf{Yes.} The positive interaction ($\beta = 0.0142$) for cancellations suggests that in concentrated markets, dominant carriers are \textit{more likely} to cancel flights, potentially exploiting market power when passengers have fewer alternatives.
    \end{itemize}
\end{frame}


\begin{frame}{Study Limitations}
    \begin{itemize}
        \item \textbf{Exogenous Shocks \& Structural Breaks}
        \begin{itemize}
            \item The 2020--2025 period contains extreme volatility (COVID-19 lockdowns and the 2022 recovery "meltdown").
            \item While fixed effects control for time, the structural relationship between HHI and delays may be fundamentally different during crisis vs. steady-state periods.
        \end{itemize}
        \addlinespace
        \item \textbf{Unobserved Environmental Variables}
        \begin{itemize}
            \item \textbf{Weather:} Lack of localized METAR data. Delays attributed to concentration may be exacerbated by localized weather patterns not fully captured by route fixed effects.
            \item \textbf{Air Traffic Control (ATC):} FAA staffing shortages during 2023--2024 acted as a system-wide supply constraint.
        \end{itemize}
        \addlinespace
        \item \textbf{Endogeneity of Market Structure}
        \begin{itemize}
            \item Airlines may exit routes \textit{because} they are prone to delays, meaning HHI could be endogenous to operational performance.
        \end{itemize}
    \end{itemize}
\end{frame}
